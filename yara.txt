YARA (Yet Another Recursive Acronym) is
an open-source, pattern-matching framework used by security professionals to identify and classify malware, malicious files, or suspicious activity based on textual or binary signatures. It acts as a "Swiss army knife" for threat intelligence, allowing analysts to create custom, flexible rules—known as YARA rules—to detect specific malicious behaviors, strings, or file characteristics within systems, memory, or network traffic.

rule Example_Linux_Ransomware_ELF {
    meta:
        description = "Detects specific behavior in Linux Ransomware ELF binaries"
        author = "CS Analyst"
        reference = "T1486 - Inhibit System Recovery"

    strings:
        // Hexadecimal string representing a specific byte sequence (no-op sled or shellcode)
        $hex_pattern = { E8 [4-8] 5D C3 } 
        
        // Strings indicating encryption and destructive terminal commands
        $enc_msg = "All your files have been encrypted" private
        $cmd_delete = "rm -rf / --no-preserve-root"
        $cmd_cron = "/etc/cron.hourly"

    condition:
        // uint32(0) == 0x464C457F verifies that it is a Linux ELF executable
        // filesize < 5MB ensures we don't waste system resources scanning massive files
        // Finally, it must contain the hex pattern and at least two of the defined strings.
        uint32(0) == 0x464C457F and filesize < 5MB and ($hex_pattern and 2 of ($enc_msg, $cmd_delete, $cmd_cron))
}

rrule complex_rule
{
    meta:
        description = "Detects hidden ActiveXObject with embedded PDF, EXE, or PNG"
        author      = "Analyst2025"
        date        = "2025-04-01"
        score       = 100

    strings:
        $js_activex = "ActiveXObject" nocase
        $pdf_magic  = { 25 50 44 46 2D }             // PDF file signature (%PDF-)
        $exe_magic  = { 4D 5A }                      // EXE file signature (MZ)
        $png_magic  = { 89 50 4E 47 0D 0A 1A 0A }   // PNG file signature

    condition:
        filesize > 200KB and $js_activex and
        any of ($pdf_magic, $exe_magic, $png_magic)
}

//SQLiteHunter.yar
rule sqlite_malware
{
    meta:
        description = "Detects suspicious SQLite malware activity"
        author      = "Analyst2025"
        score       = 100

    strings:
        // SQLite file header
        $sqlite_header = { 53 51 4C 69 74 65 20 66 6F 72 6D 61 74 20 33 00 } nocase
        // Common keywords in malware configs
        $keyword1 = "malware_config" nocase
        $keyword2 = "C2_server" nocase
        $keyword3 = "credentials" nocase

    condition:
        filesize > 50KB and $sqlite_header and any of ($keyword1, $keyword2, $keyword3)
}

//Usage
yara sqlite_malware.yar AdamFTriage.mem
yara sqlite_malware.yar windump.mem
yara sqlite_malware.yar win11dump.mem

### 🔹 Explanation

1. Meta Section

   *description → explains what this rule detects
   *score → high priority (100)

2. Strings Section

   *$sqlite_header → detects a SQLite database header (SQLite format 3)
   *$keyword1, $keyword2, $keyword3 → typical strings malware might store in SQLite for configs, credentials, or C2 info

3. Condition Section

   *filesize > 50KB → avoids small harmless SQLite files
   *$sqlite_header → ensures it’s a SQLite DB
   *any of ($keyword1, $keyword2, $keyword3) → at least one suspicious malware keyword
